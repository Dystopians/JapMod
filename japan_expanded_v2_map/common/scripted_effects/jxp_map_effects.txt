# Fresh map countries reach this helper after their history origin and daimyo
# reform are available.  It also runs on every startup/release initialization,
# so a failed first assignment is retried from the exact tag/group
# postcondition rather than being suppressed by a migration flag.
jxp_map_sync_identity_ideas_effect = {
	if = {
		limit = {
			jxp_map_is_new_daimyo_tag_trigger = yes
			jxp_is_daimyo_stage_trigger = yes
			jxp_map_has_any_origin_trigger = yes
			has_custom_ideas = no
			OR = {
				NOT = { jxp_map_has_expected_identity_ideas_trigger = yes }
				has_idea_group = jxp_map_new_daimyo_ideas
			}
		}
		swap_free_idea_group = yes
	}
}

jxp_map_initialize_effect = {
	if = {
		limit = {
			jxp_map_is_new_daimyo_tag_trigger = yes
			NOT = { has_country_flag = jxp_map_initialized }
		}
		set_country_flag = jxp_map_initialized
		jxp_map_record_origin_effect = yes
		jxp_ensure_daimyo_polity_effect = yes
	}
	# The colonial-society contract needs these northern houses to expose
	# their frontier identity before their late legacy missions complete.
	if = {
		limit = {
			OR = {
				tag = MTS
				tag = TGR
			}
		}
		jxp_a_mark_house_frontier_ready_effect = yes
	}
	jxp_map_sync_identity_ideas_effect = yes
	jxp_map_migrate_regional_flavor_v013_effect = yes
	jxp_map_migrate_identity_v012_effect = yes
	jxp_map_migrate_idea_identity_v014_effect = yes
}

# Existing v0.12 identity saves need one mission refresh to reveal the three
# regional maritime continuations.  Fresh tags defer to the identity bridge,
# avoiding a second refresh during initial setup.
jxp_map_migrate_regional_flavor_v013_effect = {
	if = {
		limit = {
			jxp_is_daimyo_stage_trigger = yes
			jxp_map_is_new_daimyo_tag_trigger = yes
			NOT = { has_country_flag = jxp_map_regional_flavor_migration_v013 }
		}
		set_country_flag = jxp_map_regional_flavor_migration_v013
		if = {
			limit = { has_country_flag = jxp_map_identity_migration_v012 }
			jxp_refresh_route_missions_effect = yes
		}
	}
}

# One-time bridge from the shared pre-identity idea group to the exact-tag
# groups and from the three-node house columns to their six-node continuations.
jxp_map_migrate_identity_v012_effect = {
	if = {
		limit = {
			jxp_is_daimyo_stage_trigger = yes
			jxp_map_is_new_daimyo_tag_trigger = yes
			NOT = { has_country_flag = jxp_map_identity_migration_v012 }
		}
		set_country_flag = jxp_map_identity_migration_v012
		set_country_flag = jxp_map_runtime_migration_v011
		jxp_map_sync_identity_ideas_effect = yes
		jxp_refresh_route_missions_effect = yes
	}
	else = { jxp_map_reconcile_daimyo_missions_effect = yes }
}

# 0.14 makes identity synchronization idempotent for fresh map tags while
# retaining one forced cleanup after a map-origin country forms a route tag.
# Treat the version marker as a transaction commit: an old save carrying the
# marker but not the postcondition clears it, retries, and commits it only after
# the exact identity is restored (or an explicit custom/unmanaged opt-out).
jxp_map_migrate_idea_identity_v014_effect = {
	if = {
		limit = {
			jxp_map_has_any_origin_trigger = yes
			OR = {
				NOT = { has_country_flag = jxp_map_idea_identity_migration_v014 }
				NOT = { jxp_map_idea_identity_v014_postcondition_trigger = yes }
			}
		}
		clr_country_flag = jxp_map_idea_identity_migration_v014
		if = {
			limit = {
				jxp_map_is_new_daimyo_tag_trigger = yes
			}
			jxp_map_sync_identity_ideas_effect = yes
		}
		else_if = {
			limit = {
				has_custom_ideas = no
				jxp_uses_route_national_ideas_trigger = yes
			}
			if = {
				limit = { jxp_map_has_any_identity_idea_group_trigger = yes }
				jxp_force_sync_route_national_ideas_effect = yes
			}
			else = { jxp_sync_route_national_ideas_effect = yes }
		}
		if = {
			limit = { jxp_map_idea_identity_v014_postcondition_trigger = yes }
			set_country_flag = jxp_map_idea_identity_migration_v014
		}
	}
}
