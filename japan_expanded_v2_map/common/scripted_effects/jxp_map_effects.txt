jxp_map_initialize_geography_contract_effect = {
	if = {
		limit = { NOT = { has_global_flag = jxp_map_geography_contract_v011 } }
		set_global_flag = jxp_map_geography_contract_v011
		every_province = {
			limit = { area = jxp_western_kyushu_area }
			set_province_flag = jxp_map_compat_shimabara_belt
			set_province_flag = jxp_map_compat_wokou_waters
		}
		every_province = {
			limit = { area = jxp_western_saigoku_area }
			set_province_flag = jxp_map_compat_ikko_heartland
			set_province_flag = jxp_map_compat_setouchi
			set_province_flag = jxp_map_compat_wokou_waters
		}
		every_province = {
			limit = { area = jxp_ise_kii_area }
			set_province_flag = jxp_map_compat_ikko_heartland
		}
		every_province = {
			limit = { area = jxp_ryukyu_area }
			set_province_flag = jxp_map_compat_wokou_waters
		}
	}
}

jxp_map_initialize_effect = {
	if = {
		limit = {
			jxp_map_is_new_daimyo_tag_trigger = yes
			NOT = { has_country_flag = jxp_map_initialized }
		}
		set_country_flag = jxp_map_initialized
		jxp_map_record_origin_effect = yes
		jxp_ensure_daimyo_polity_effect = yes
	}
	jxp_map_migrate_runtime_v011_effect = yes
}

jxp_map_migrate_runtime_v011_effect = {
	if = {
		limit = {
			jxp_is_daimyo_stage_trigger = yes
			jxp_map_has_any_origin_trigger = yes
			NOT = { has_country_flag = jxp_map_runtime_migration_v011 }
		}
		set_country_flag = jxp_map_runtime_migration_v011
		jxp_refresh_route_missions_effect = yes
	}
	else = { jxp_map_reconcile_daimyo_missions_effect = yes }
}

jxp_map_reconcile_daimyo_missions_effect = {
	if = {
		limit = { jxp_map_daimyo_mission_tree_needs_reconcile_trigger = yes }
		jxp_refresh_route_missions_effect = yes
	}
}
