# The main mod validates shared slots 1/2/4/5. The companion owns the exact
# slot-3 fingerprint so the main mod never has to reference optional missions.
jxp_map_daimyo_mission_fingerprint_valid_trigger = {
	jxp_is_daimyo_stage_trigger = yes
	jxp_map_has_any_origin_trigger = yes
	NOT = {
		OR = {
			has_mission = jxp_mission_house_warrior_muster
			has_mission = jxp_mission_house_court_genealogies
			has_mission = jxp_mission_house_sea_port_registers
			has_mission = jxp_mission_house_frontier_passes
			has_mission = jxp_mission_house_temple_ledgers
		}
	}
	OR = {
		AND = {
			jxp_map_has_warrior_origin_trigger = yes
			has_mission = jxp_map_mission_warrior_muster
			NOT = {
				OR = {
					has_mission = jxp_map_mission_court_genealogies
					has_mission = jxp_map_mission_sea_port_registers
					has_mission = jxp_map_mission_frontier_passes
					has_mission = jxp_map_mission_temple_ledgers
				}
			}
		}
		AND = {
			jxp_map_has_court_origin_trigger = yes
			has_mission = jxp_map_mission_court_genealogies
			NOT = {
				OR = {
					has_mission = jxp_map_mission_warrior_muster
					has_mission = jxp_map_mission_sea_port_registers
					has_mission = jxp_map_mission_frontier_passes
					has_mission = jxp_map_mission_temple_ledgers
				}
			}
		}
		AND = {
			jxp_map_has_sea_origin_trigger = yes
			has_mission = jxp_map_mission_sea_port_registers
			NOT = {
				OR = {
					has_mission = jxp_map_mission_warrior_muster
					has_mission = jxp_map_mission_court_genealogies
					has_mission = jxp_map_mission_frontier_passes
					has_mission = jxp_map_mission_temple_ledgers
				}
			}
		}
		AND = {
			jxp_map_has_frontier_origin_trigger = yes
			has_mission = jxp_map_mission_frontier_passes
			NOT = {
				OR = {
					has_mission = jxp_map_mission_warrior_muster
					has_mission = jxp_map_mission_court_genealogies
					has_mission = jxp_map_mission_sea_port_registers
					has_mission = jxp_map_mission_temple_ledgers
				}
			}
		}
		AND = {
			jxp_map_has_temple_market_origin_trigger = yes
			has_mission = jxp_map_mission_temple_ledgers
			NOT = {
				OR = {
					has_mission = jxp_map_mission_warrior_muster
					has_mission = jxp_map_mission_court_genealogies
					has_mission = jxp_map_mission_sea_port_registers
					has_mission = jxp_map_mission_frontier_passes
				}
			}
		}
	}
}
