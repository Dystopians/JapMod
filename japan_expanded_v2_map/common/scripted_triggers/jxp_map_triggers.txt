# Transaction postcondition for the v0.14 idea migration.  Custom ideas and
# countries outside the maintained map/route identities are explicit opt-outs;
# maintained identities must hold their exact current group, and route tags may
# not retain a serialized map-daimyo group underneath it.
jxp_map_idea_identity_v014_postcondition_trigger = {
	OR = {
		has_custom_ideas = yes
		AND = {
			jxp_map_is_new_daimyo_tag_trigger = yes
			jxp_map_has_expected_identity_ideas_trigger = yes
			NOT = { has_idea_group = jxp_map_new_daimyo_ideas }
		}
		AND = {
			jxp_uses_route_national_ideas_trigger = yes
			jxp_has_expected_route_national_ideas_trigger = yes
			NOT = { jxp_map_has_any_identity_idea_group_trigger = yes }
		}
		AND = {
			NOT = { jxp_map_is_new_daimyo_tag_trigger = yes }
			NOT = { jxp_uses_route_national_ideas_trigger = yes }
		}
	}
}

# slot-3 fingerprint so the main mod never has to reference optional missions.
jxp_map_daimyo_mission_fingerprint_valid_trigger = {
	jxp_is_daimyo_stage_trigger = yes
	jxp_map_has_any_origin_trigger = yes
	NOT = {
		OR = {
			has_mission = jxp_mission_house_warrior_muster
			has_mission = jxp_mission_house_court_genealogies
			has_mission = jxp_mission_house_sea_port_registers
			has_mission = jxp_mission_house_frontier_passes
			has_mission = jxp_mission_house_temple_ledgers
		}
	}
	OR = {
		jxp_map_major_daimyo_mission_fingerprint_valid_trigger = yes
		AND = {
			NOT = { jxp_map_is_major_daimyo_identity_tag_trigger = yes }
			jxp_map_has_warrior_origin_trigger = yes
			has_mission = jxp_map_mission_warrior_muster
			NOT = {
				OR = {
					has_mission = jxp_map_mission_court_genealogies
					has_mission = jxp_map_mission_sea_port_registers
					has_mission = jxp_map_mission_frontier_passes

			has_mission = jxp_map_mission_court_genealogies
