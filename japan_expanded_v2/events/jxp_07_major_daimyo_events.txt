country_event = {
	id = jxp_major_daimyo.1
	title = "jxp_major_daimyo.1.t"
	desc = "jxp_major_daimyo.1.d"
	picture = LAND_MILITARY_eventPicture

	trigger = {
		jxp_is_daimyo_stage_trigger = yes
		tag = ODA
		is_year = 1460
		NOT = { has_country_flag = jxp_major_daimyo_oda_seen }
	}

	mean_time_to_happen = { months = 140 }

	option = {
		name = "jxp_major_daimyo.1.a"
		set_country_flag = jxp_major_daimyo_oda_seen
		jxp_ensure_daimyo_polity_effect = yes
		jxp_add_tenka_order_5_effect = yes
		add_country_modifier = {
			name = jxp_42_oda_teppo_muster
			duration = 3650
		}
		add_mil_power = 25
	}
	option = {
		name = "jxp_major_daimyo.1.b"
		set_country_flag = jxp_major_daimyo_oda_seen
		jxp_ensure_daimyo_polity_effect = yes
		jxp_add_imperial_sanction_5_effect = yes
		add_country_modifier = {

		}
		add_prestige = 5
	}
}
