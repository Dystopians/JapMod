namespace = jxp_confucian_incident

# Confucian Japan cannot use the Shinto isolationism incident interface after
# conversion. These deliberately sparse chains preserve the historical themes
# and feed their choices into JXP's persistent polity values instead.

country_event = {
	id = jxp_confucian_incident.98
	title = none
	desc = none
	picture = COURT_eventPicture
	hidden = yes

	trigger = {
		jxp_confucian_shinto_bridge_country_trigger = yes
		NOT = { has_country_flag = jxp_confucian_bridge_incident_state_v0231 }
	}

	mean_time_to_happen = { days = 1 }

	immediate = {
		set_country_flag = jxp_confucian_bridge_incident_state_v0231
		if = {
			limit = { has_country_flag = jxp_confucian_bridge_neo_done }
			set_country_flag = end_of_neo_confucianism_chain
		}
		if = {
			limit = { has_country_flag = jxp_confucian_bridge_urban_done }
			set_country_flag = end_of_urbanisation_chain
		}
		if = {
			limit = { has_country_flag = jxp_confucian_bridge_nanban_done }
			set_country_flag = end_of_nanban_chain
			if = {
				limit = {
					NOT = {
						OR = {
							has_country_flag = jap_chose_kaikoku_flag
							has_country_flag = jap_chose_trade_with_west_flag
							has_country_flag = jap_chose_sakoku_flag
						}
					}
				}
				set_country_flag = jap_chose_trade_with_west_flag
			}
		}
		if = {
			limit = { has_country_flag = jxp_confucian_bridge_firearms_done }
			set_country_flag = end_of_incident_firearms_chain
			if = {
				limit = { has_country_modifier = jxp_confucian_standard_calibers }
				clr_country_flag = proliferation_firearms_flag
				clr_country_flag = example_of_bushido_flag
				set_country_flag = perfecting_the_musket_flag
			}
			else_if = {
				limit = { has_country_modifier = jxp_confucian_guarded_arsenals }
				clr_country_flag = perfecting_the_musket_flag
				clr_country_flag = proliferation_firearms_flag
				set_country_flag = example_of_bushido_flag
			}
			else_if = {
				limit = {
					NOT = {
						OR = {
							has_country_flag = perfecting_the_musket_flag
							has_country_flag = proliferation_firearms_flag
							has_country_flag = example_of_bushido_flag
						}
					}
				}
				set_country_flag = proliferation_firearms_flag
			}
		}
		if = {
			limit = { has_country_flag = jxp_confucian_bridge_christian_done }
			set_country_flag = end_of_incident_spread_of_christianity_chain
			if = {
				limit = {
					NOT = {
						OR = {
							has_country_flag = significant_christian_presence_flag
							has_country_flag = christianity_defeated_flag
						}
					}
				}
				set_country_flag = significant_christian_presence_flag
			}
		}
		if = {
			limit = { has_country_flag = jxp_confucian_bridge_authority_done }
			set_country_flag = end_of_incident_shogunate_authority_chain
		}
		if = {
			limit = { has_country_flag = jxp_confucian_bridge_ikko_done }
			set_country_flag = end_of_ikko_chain
		}
		if = {
			limit = { has_country_flag = jxp_confucian_bridge_wokou_done }
			set_country_flag = end_of_wokou_chain
		}
	}

	option = { name = "OK" }
}
