	id = jxp_colonial_society.2
	title = none
	desc = none
	picture = COLONIZATION_eventPicture
	hidden = yes
	is_triggered_only = yes

	immediate = {
		clr_country_flag = jxp_b_colonial_pulse_scheduled
		if = {
			limit = { jxp_b_91_colonial_society_active_trigger = yes }
			jxp_b_91_clamp_colonial_variables_effect = yes
			jxp_b_91_sync_colonial_stage_effect = yes
			if = {
				limit = { jxp_b_96_colonial_crisis_ready_trigger = yes }
				country_event = { id = jxp_colonial_independence.1 days = 30 }
			}
			if = {
				limit = { NOT = { has_country_modifier = jxp_b_93_society_event_recess } }
				add_country_modifier = { name = jxp_b_93_society_event_recess duration = 730 }
				country_event = { id = jxp_colonial_society.90 days = 1 }
			}
			jxp_b_91_schedule_colonial_pulse_effect = yes
		}
		else = { jxp_b_91_colonial_society_cleanup_effect = yes }
	}

	option = { name = "OK" }
}
