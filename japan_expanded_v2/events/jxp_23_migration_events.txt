country_event = {
	id = jxp_migration_v023.5
	title = none
	desc = none
	picture = COURT_eventPicture
	hidden = yes

	trigger = {
		jxp_is_japanese_polity_trigger = yes
		NOT = { has_country_flag = jxp_mission_expansion_migration_v024 }
	}

	mean_time_to_happen = { days = 1 }

	immediate = {
		set_country_flag = jxp_mission_expansion_migration_v024
		jxp_sanitize_route_state_effect = yes
		if = {
			limit = {
				jxp_is_daimyo_stage_trigger = yes
				NOT = { has_country_flag = jxp_24_house_path_completed }
				OR = {
					mission_completed = jxp_mission_house_warrior_banner_legacy
					mission_completed = jxp_mission_house_court_kanrei_memory
					mission_completed = jxp_mission_house_sea_strait_laws
					mission_completed = jxp_mission_house_frontier_border_oaths
					mission_completed = jxp_mission_house_temple_public_law
				}
			}
			set_country_flag = jxp_24_house_path_completed
		}

		if = {
			limit = {
				jxp_is_daimyo_stage_trigger = yes
				jxp_has_warrior_house_origin_trigger = yes
				has_country_flag = jxp_25_warrior_agenda_seen
				NOT = {
					OR = {
						has_country_flag = jxp_24_warrior_agenda_rolls
						has_country_flag = jxp_24_warrior_agenda_captains
					}
				}
			}
			set_country_flag = jxp_24_warrior_agenda_rolls
		}
		if = {
			limit = {
				jxp_is_daimyo_stage_trigger = yes
				jxp_has_court_house_origin_trigger = yes
				has_country_flag = jxp_25_court_agenda_seen
				NOT = {
					OR = {
						has_country_flag = jxp_24_court_agenda_ranks
						has_country_flag = jxp_24_court_agenda_stewards
					}
				}
			}
			set_country_flag = jxp_24_court_agenda_ranks
		}
		if = {
			limit = {
				jxp_is_daimyo_stage_trigger = yes
				jxp_has_maritime_house_origin_trigger = yes
				has_country_flag = jxp_25_maritime_agenda_seen
				NOT = {
					OR = {
						has_country_flag = jxp_24_maritime_agenda_merchants
						has_country_flag = jxp_24_maritime_agenda_ship_rolls
					}
				}
			}
			set_country_flag = jxp_24_maritime_agenda_merchants
		}
		if = {
			limit = {
				jxp_is_daimyo_stage_trigger = yes
				jxp_has_frontier_house_origin_trigger = yes
				has_country_flag = jxp_25_frontier_agenda_seen
				NOT = {
					OR = {
						has_country_flag = jxp_24_frontier_agenda_wardens
						has_country_flag = jxp_24_frontier_agenda_brokers
					}
				}
			}
			set_country_flag = jxp_24_frontier_agenda_wardens
		}
		if = {
			limit = {
				jxp_is_daimyo_stage_trigger = yes
				jxp_has_temple_market_house_origin_trigger = yes
				has_country_flag = jxp_25_temple_market_agenda_seen
				NOT = {
					OR = {
						has_country_flag = jxp_24_temple_agenda_privileges
						has_country_flag = jxp_24_temple_agenda_ledgers
					}
				}
			}
			set_country_flag = jxp_24_temple_agenda_privileges
		}

		if = {
			limit = {
				jxp_is_daimyo_stage_trigger = yes
				jxp_has_warrior_house_origin_trigger = yes
				NOT = { has_country_flag = jxp_25_warrior_agenda_seen }

				NOT = { has_country_flag = jxp_25_maritime_agenda_seen }
