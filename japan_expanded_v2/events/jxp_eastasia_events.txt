country_event = {
	id = jxp_eastasia.1
	title = "jxp_eastasia.1.t"
	desc = "jxp_eastasia.1.d"
	picture = DIPLOMACY_eventPicture

	trigger = {
		jxp_is_japanese_polity_trigger = yes
		has_country_flag = jxp_polity_initialized
		is_year = 1500
		NOT = { has_country_flag = jxp_korean_embassy_seen }
		jxp_oceanic_opening_at_least_20 = yes
		num_of_ports = 3
		OR = {
			korea_region = { has_discovered = ROOT }
			any_known_country = {
				capital_scope = { region = korea_region }
			}
			num_of_provinces_owned_or_owned_by_non_sovereign_subjects_with = {
				value = 1
				region = korea_region
				is_city = yes
			}
		}
	}

	mean_time_to_happen = {
		months = 160
	}

	option = {
		name = "jxp_eastasia.1.a"
		set_country_flag = jxp_korean_embassy_seen
		jxp_add_imperial_sanction_5_effect = yes
		add_prestige = 5
		add_country_modifier = {
			name = jxp_celestial_diplomacy
			duration = 3650
		}
	}

	option = {
		name = "jxp_eastasia.1.b"
		set_country_flag = jxp_korean_embassy_seen
		jxp_add_oceanic_opening_5_effect = yes
		add_navy_tradition = 5
		add_country_modifier = {
			name = jxp_manchu_korean_route
			duration = 3650
		}
	}
}
