namespace = jxp_route_house_compromise

country_event = {
	id = jxp_route_house_compromise.1
	title = "jxp_route_house_compromise.1.t"
	desc = "jxp_route_house_compromise.1.d"
	picture = COURT_eventPicture
	is_triggered_only = yes
	trigger = {
		jxp_is_unified_japan_state_trigger = yes
		has_country_flag = jxp_path_sakoku
		NOT = { has_country_flag = jxp_29_route_house_compromise_taken }
	}
	option = {
		name = "jxp_route_house_compromise.1.a"
		set_country_flag = jxp_29_route_house_compromise_taken
		set_country_flag = jxp_29_sakoku_house_compromise_seen
		jxp_add_tenka_order_5_effect = yes
		jxp_subtract_oceanic_opening_5_effect = yes
		add_country_modifier = { name = jxp_29_sakoku_house_ordinance duration = 5475 }
	}
	option = {
		name = "jxp_route_house_compromise.1.b"
		set_country_flag = jxp_29_route_house_compromise_taken
		set_country_flag = jxp_29_sakoku_house_compromise_seen
		jxp_apply_founder_house_compromise_effect = yes
	}
}
