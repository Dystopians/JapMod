country_event = {
	id = jxp_minor_daimyo.1
	title = "jxp_minor_daimyo.1.t"
	desc = "jxp_minor_daimyo.1.d"
	picture = COURT_eventPicture

	trigger = {
		jxp_is_daimyo_stage_trigger = yes
		OR = {
			tag = HSK
			tag = YMN
			tag = AKM
			tag = SBA
			tag = HTK
			tag = IKE
		}
		NOT = { has_country_flag = jxp_minor_daimyo_kinai_houses_seen }
	}

	mean_time_to_happen = { months = 240 }

	option = {
		name = "jxp_minor_daimyo.1.a"
		set_country_flag = jxp_minor_daimyo_kinai_houses_seen
		jxp_ensure_daimyo_polity_effect = yes
		jxp_add_imperial_sanction_5_effect = yes
		add_country_modifier = {
			name = jxp_42_kinai_old_letters
			duration = 3650
		}
		add_prestige = 5
	}
	option = {
		name = "jxp_minor_daimyo.1.b"
		set_country_flag = jxp_minor_daimyo_kinai_houses_seen
		jxp_ensure_daimyo_polity_effect = yes
		jxp_add_tenka_order_5_effect = yes
		add_country_modifier = {
			name = jxp_42_kinai_domain_ledgers
			duration = 3650
		}
		add_adm_power = 25
	}
}
