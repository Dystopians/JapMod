country_event = {
	id = jxp_route_house_resonance.1
	title = "jxp_route_house_resonance.1.t"
	desc = "jxp_route_house_resonance.1.d"
	picture = LAND_MILITARY_eventPicture
	trigger = {
		jxp_is_unified_japan_state_trigger = yes
		jxp_has_warrior_house_origin_trigger = yes
		OR = {
			jxp_has_any_route_trigger = yes
			tag = KJP
			tag = CJP
			tag = EJP
			tag = RFJ
			tag = SJP
			tag = IJP
			tag = WAK
		}
		NOT = { has_country_flag = jxp_30_warrior_route_resonance_seen }
	}
	mean_time_to_happen = {
		months = 260
		modifier = {
			factor = 0.75
			OR = {
				has_country_flag = jxp_path_sakoku
				has_country_flag = jxp_path_ikko
				tag = IJP
			}
		}
	}
	option = {
		name = "jxp_route_house_resonance.1.a"
		set_country_flag = jxp_30_warrior_route_resonance_seen
		jxp_add_current_route_resonance_effect = yes
		add_country_modifier = { name = jxp_30_route_over_warrior_houses duration = 3650 }
	}
	option = {
		name = "jxp_route_house_resonance.1.b"
		set_country_flag = jxp_30_warrior_route_resonance_seen
		jxp_add_current_house_resonance_effect = yes
		add_country_modifier = { name = jxp_30_warrior_houses_temper_route duration = 3650 }
	}
}
