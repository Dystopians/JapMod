country_event = {
	id = jxp_confucian.3
	title = none
	desc = none
	picture = COURT_eventPicture
	hidden = yes

	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				jxp_confucian_route_country_trigger = yes
				NOT = { has_country_flag = jxp_confucian_native_centers_v1 }
			}
			jxp_spawn_confucian_reform_centers_effect = yes
		}
		if = {
			limit = {
				any_province = {
					region = japan_region
					religion = shinto
					NOT = { province_id = 4359 }
				}
			}
			country_event = { id = jxp_confucian.3 days = 365 }
		}
		else = {
			clr_country_flag = jxp_confucian_conversion_active
			set_country_flag = jxp_confucian_order_complete
			if = {
				limit = { NOT = { has_country_flag = jxp_confucian_completion_rewarded } }
				country_event = { id = jxp_confucian.4 days = 30 }
			}
		}
	}

	option = {
		name = "OK"
	}
}
