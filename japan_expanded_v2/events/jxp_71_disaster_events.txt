# Scripted effects load before custom disaster IDs are safe to validate in
# EU4 1.37.5. Debug controls dispatch here so the engine resolves the target
# disaster from the event layer, matching the vanilla add-progress pattern.

	title = none
	desc = none
	picture = COURT_eventPicture
	hidden = yes
	is_triggered_only = yes

		}

# Disaster hooks accept an event ID, not an inline effect block. Keep all
# teardown state in hidden triggered-only events so every disaster registers.
country_event = {
	id = jxp_disasters.90
	title = none
	desc = none
	picture = COURT_eventPicture
	hidden = yes
	is_triggered_only = yes

	immediate = {
		set_country_flag = jxp_shimabara_resolved
		clr_country_flag = jxp_shimabara_crisis_active
		clr_country_flag = jxp_shimabara_mitigated
		remove_country_modifier = jxp_shimabara_mediation
		remove_country_modifier = jxp_shimabara_suppressed
	}

	option = { name = "OK" }
}
