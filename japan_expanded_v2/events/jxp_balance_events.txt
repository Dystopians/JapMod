namespace = jxp_balance

country_event = {
	id = jxp_balance.1
	title = "jxp_balance.1.t"
	desc = "jxp_balance.1.d"
	picture = COURT_eventPicture

	trigger = {
		jxp_is_japanese_polity_trigger = yes
		has_country_flag = jxp_polity_initialized
		has_any_disaster = no
		is_at_war = no
		jxp_tenka_order_at_least_75 = yes
		NOT = { has_country_flag = jxp_balance_tenka_high_cooldown }
	}

	mean_time_to_happen = {
		months = 240
		modifier = {
			factor = 0.75
			has_country_flag = jxp_path_sakoku
		}
		modifier = {
			factor = 0.75
			has_country_flag = jxp_path_imperial
		}
	}

	option = {
		name = "jxp_balance.1.a"
		hidden_effect = {
			set_country_flag = jxp_balance_tenka_high_cooldown
			country_event = { id = jxp_balance.101 days = 7300 }
		}
		jxp_subtract_tenka_order_5_effect = yes
		add_prestige = 5
	}

	option = {
		name = "jxp_balance.1.b"
		hidden_effect = {
			set_country_flag = jxp_balance_tenka_high_cooldown
			country_event = { id = jxp_balance.101 days = 7300 }
		}
		jxp_add_tenka_order_5_effect = yes
		add_adm_power = -25
		add_legitimacy = 5
	}
}
