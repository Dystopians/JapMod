	id = jxp_namban.1
	title = "jxp_namban.1.t"
	desc = "jxp_namban.1.d"
	picture = COURT_eventPicture

	trigger = {
		jxp_is_japanese_polity_trigger = yes
		is_year = 1543
		NOT = { has_country_flag = jxp_tanegashima_seen }
		any_owned_province = {
			region = japan_region
			has_port = yes
		}
	}

	mean_time_to_happen = {
		months = 120
	}

	option = {
		name = "jxp_namban.1.a"
		set_country_flag = jxp_tanegashima_seen
		jxp_add_oceanic_opening_5_effect = yes
		add_country_modifier = {
			name = jxp_tanegashima_foundries
			duration = 7300
		}
	}
	option = {
		name = "jxp_namban.1.b"
		set_country_flag = jxp_tanegashima_seen
		jxp_subtract_oceanic_opening_5_effect = yes
		add_mil_power = 50
	}
}
