# Each visible council is both a low-frequency natural event and the endpoint
# of its route's player decision.  Every selected reform owns an independent
# option with a real era-attribute gain/cost and a finite ten-year result.

	id = jxp_route_parity_four.1
	title = "jxp_route_parity_four.1.t"
	desc = "jxp_route_parity_four.1.d"
	picture = COURT_eventPicture

	trigger = {
		jxp_75_is_kirishitan_route_trigger = yes
		jxp_75_has_kirishitan_policy_reform_trigger = yes
		NOT = { has_country_modifier = jxp_75_route_policy_recent }
	}

	mean_time_to_happen = { months = 120 }

	option = {
		name = "jxp_route_parity_four.1.a"
		trigger = {
			jxp_75_is_kirishitan_route_trigger = yes
			has_reform = jxp_reform_kirishitan_seminaries
		}
		add_country_modifier = {
			name = jxp_75_route_policy_recent
			duration = 3650
		}
		add_country_modifier = {
			name = jxp_75_kirishitan_seminary_visitation
			duration = 3650
		}
		jxp_add_oceanic_opening_5_effect = yes
		jxp_subtract_tenka_order_5_effect = yes

			factor = 1
			modifier = { factor = 2 NOT = { jxp_oceanic_opening_at_least_50 = yes } }

# Defensive fallback for a route change that bypasses the canonical cleanup
# path.  The public reset effect still gives route/debug roots an immediate
# cleanup hook; this hidden event bounds any stale state to roughly one day.
country_event = {
	id = jxp_route_parity_four.100
	title = none
	desc = none
	picture = COURT_eventPicture
	hidden = yes

	trigger = {
		OR = {
			AND = {
				OR = {
					has_country_modifier = jxp_75_kirishitan_seminary_visitation
					has_country_modifier = jxp_75_kirishitan_mercy_accounts
					has_country_modifier = jxp_75_kirishitan_harbor_chaplains
				}
				NOT = { jxp_75_is_kirishitan_route_trigger = yes }
			}
			AND = {
				OR = {
					has_country_modifier = jxp_75_confucian_examination_rosters
					has_country_modifier = jxp_75_confucian_code_commission
					has_country_modifier = jxp_75_confucian_three_teachings_concordance
				}
				NOT = { jxp_75_is_confucian_route_trigger = yes }
			}
			AND = {
				OR = {
					has_country_modifier = jxp_75_reformed_printing_consistory
					has_country_modifier = jxp_75_reformed_civic_auditors
					has_country_modifier = jxp_75_reformed_fleet_articles
				}
				NOT = { jxp_75_is_reformed_route_trigger = yes }
			}
			AND = {
				OR = {
					has_country_modifier = jxp_75_kaikyo_monsoon_accounts
					has_country_modifier = jxp_75_kaikyo_endowment_granaries
					has_country_modifier = jxp_75_kaikyo_convoy_muster
				}
				NOT = { jxp_75_is_kaikyo_route_trigger = yes }
			}
			AND = {
				has_country_modifier = jxp_75_route_policy_recent
				NOT = {
					OR = {
						jxp_75_is_kirishitan_route_trigger = yes
						jxp_75_is_confucian_route_trigger = yes
						jxp_75_is_reformed_route_trigger = yes
						jxp_75_is_kaikyo_route_trigger = yes
					}
				}
			}
		}
	}

	mean_time_to_happen = { days = 1 }

	immediate = {
		jxp_75_reset_route_parity_four_effect = yes
	}

	option = { name = "OK" }
}
