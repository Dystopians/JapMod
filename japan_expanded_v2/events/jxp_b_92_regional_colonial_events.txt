	title = "jxp_b_regional_colonial.1.t"
	desc = "jxp_b_regional_colonial.1.d"

	is_triggered_only = yes

	option = {
		name = "jxp_b_regional_colonial.1.a"
		add_treasury = 100
		add_country_modifier = {
			name = jxp_b_92_south_seas_host_charter_rents
			duration = 3650
		}
		random_owned_province = {
			limit = {
				jxp_b_92_southeast_asian_port_province_trigger = yes
				NOT = { has_province_modifier = jxp_b_92_authorized_japan_town_port }
			}
			add_province_modifier = {
				name = jxp_b_92_authorized_japan_town_port
				duration = -1
			}
		}
		custom_tooltip = jxp_b_92_host_grant_charter_tt
		hidden_effect = {
			FROM = {
				jxp_b_92_accept_authorized_japan_town_effect = yes
				country_event = { id = jxp_b_regional_colonial.9 days = 1 }
			}
		}
		ai_chance = {
			factor = 2
			modifier = { factor = 2 has_opinion = { who = FROM value = 125 } }
			modifier = { factor = 1.5 reverse_has_opinion = { who = FROM value = 75 } }
			modifier = { factor = 0.5 war_exhaustion = 5 }
		}
	}

	option = {
		name = "jxp_b_regional_colonial.1.b"
		add_prestige = 5
		custom_tooltip = jxp_b_92_host_refuse_charter_tt
		hidden_effect = {
			FROM = {
				clr_country_flag = jxp_b_92_south_seas_negotiation_pending
				add_dip_power = 25
				add_prestige = -5
				add_country_modifier = {
					name = jxp_b_92_south_seas_negotiation_rebuffed
					duration = 1825
				}
				country_event = { id = jxp_b_regional_colonial.10 days = 1 }
			}
		}
		ai_chance = {
			factor = 0.5
			modifier = { factor = 3 NOT = { stability = 0 } }
			modifier = { factor = 2 war_exhaustion = 5 }
		}
	}
}
