# Readiness and war-end monitors are self-scheduled only for the prepared country.
country_event = {
	id = jxp_korea_campaign.940
	title = "jxp_korea_campaign.940.t"
	desc = "jxp_korea_campaign.940.d"
	picture = DIPLOMACY_eventPicture
	hidden = yes
	is_triggered_only = yes

	trigger = { has_country_flag = jxp_b_101_korea_campaign_active }

	option = {
		name = "jxp_korea_campaign.940.a"
		clr_country_flag = jxp_b_101_korea_monitor_scheduled
		if = {
			limit = {
				war_with = KOR
				NOT = { has_country_flag = jxp_b_101_korea_war_started }
			}
			set_country_flag = jxp_b_101_korea_war_started
			country_event = { id = jxp_korea_campaign.100 days = 1 }
			country_event = { id = jxp_korea_campaign.101 days = 45 }
			country_event = { id = jxp_korea_campaign.941 days = 90 }
		}
		else_if = {
			limit = { has_country_flag = jxp_b_101_korea_war_started }
			country_event = { id = jxp_korea_campaign.941 days = 30 }
		}
		else_if = {
			limit = { NOT = { has_country_modifier = jxp_b_101_korea_expedition_ready } }
			jxp_b_101_korea_campaign_active_cleanup_effect = yes
		}
		else = {
			set_country_flag = jxp_b_101_korea_monitor_scheduled
			country_event = { id = jxp_korea_campaign.940 days = 60 }
		}
	}
}

	id = jxp_korea_campaign.114
	title = "jxp_korea_campaign.114.t"
	desc = "jxp_korea_campaign.114.d"
	picture = jxp_b_korea_campaign_eventPicture
	is_triggered_only = yes
	trigger = { jxp_b_101_korea_war_active_trigger = yes NOT = { has_country_flag = jxp_b_101_korea_event_114_seen } }
	immediate = { set_country_flag = jxp_b_101_korea_event_114_seen }
	option = {
		name = "jxp_korea_campaign.114.a"
		trigger = { ruler_age = 60 }
		ai_chance = { factor = 100 }
		set_country_flag = jxp_b_101_korea_old_ruler_died
		kill_ruler = yes
		add_stability = -1
		add_country_modifier = { name = jxp_b_101_succession_shock duration = 3650 }
		hidden_effect = {
			change_variable = { which = jxp_b_korea_daimyo_compliance value = -15 }
			jxp_b_101_clamp_korea_preparation_variables_effect = yes
		}
	}
	option = {
		name = "jxp_korea_campaign.114.b"
		trigger = { NOT = { ruler_age = 60 } }
		ai_chance = { factor = 100 }
		add_army_tradition = -10
		add_country_modifier = { name = jxp_b_101_succession_shock duration = 1825 }
		hidden_effect = {
			change_variable = { which = jxp_b_korea_expedition_support value = -10 }

		}
