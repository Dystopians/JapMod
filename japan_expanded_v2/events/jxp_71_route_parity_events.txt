namespace = jxp_route_parity

# Each council can arise organically or be convened through its route decision.
# A ten-year self-expiring modifier is both the recurrence lock and the visible
# lifecycle marker; no persistent country flag is created by this system.
country_event = {
	id = jxp_route_parity.1
	title = "jxp_route_parity.1.t"
	desc = "jxp_route_parity.1.d"
	picture = COURT_eventPicture

	trigger = {
		jxp_71_is_sakoku_route_trigger = yes
		jxp_71_has_sakoku_council_reform_trigger = yes
		NOT = { has_country_modifier = jxp_71_route_council_recent }
	}

	mean_time_to_happen = {
		months = 96
	}

	option = {
		name = "jxp_route_parity.1.a"
		trigger = { has_reform = jxp_reform_sakoku_sankin_kotai_roads }
		add_country_modifier = {
			name = jxp_71_route_council_recent
			duration = 3650
		}
		add_country_modifier = {
			name = jxp_71_sakoku_post_road_registers
			duration = 3650
		}
		jxp_add_tenka_order_5_effect = yes
		jxp_add_imperial_sanction_5_effect = yes
		jxp_subtract_oceanic_opening_5_effect = yes
		ai_chance = {
			factor = 1
			modifier = { factor = 2 NOT = { jxp_tenka_order_at_least_50 = yes } }
		}
	}
	option = {
		name = "jxp_route_parity.1.b"
		trigger = { has_reform = jxp_reform_sakoku_coastal_barriers }
		add_country_modifier = {
			name = jxp_71_route_council_recent
			duration = 3650
		}
		add_country_modifier = {
			name = jxp_71_sakoku_coastal_musters
			duration = 3650
		}
		jxp_add_tenka_order_5_effect = yes
		jxp_subtract_oceanic_opening_5_effect = yes
		ai_chance = {
			factor = 1
			modifier = { factor = 2 jxp_oceanic_opening_at_least_50 = yes }
		}
	}
	option = {
		name = "jxp_route_parity.1.c"
		trigger = { has_reform = jxp_reform_sakoku_hidden_learning }
		add_country_modifier = {
			name = jxp_71_route_council_recent
			duration = 3650
		}
		add_country_modifier = {
			name = jxp_71_sakoku_sealed_translation_room
			duration = 3650
		}
		jxp_add_oceanic_opening_5_effect = yes
		jxp_subtract_tenka_order_5_effect = yes
		ai_chance = {
			factor = 1
			modifier = { factor = 2 NOT = { jxp_oceanic_opening_at_least_35 = yes } }
		}
	}
}
