country_event = {
	id = jxp_ikko.2
	title = "jxp_ikko.2.t"
	desc = "jxp_ikko.2.d"
	picture = COURT_eventPicture

	trigger = {
		jxp_is_japanese_polity_trigger = yes
		has_country_flag = jxp_path_ikko
		NOT = { has_country_flag = jxp_ikko_terauchi_seen }
	}

	mean_time_to_happen = {
		days = 1800
	}

	option = {
		name = "jxp_ikko.2.a"
		set_country_flag = jxp_ikko_terauchi_seen
		set_country_flag = jxp_ikko_contact
		if = {
			limit = { owns = 1021 }
			1021 = {
				add_province_modifier = {
					name = jxp_ikko_terauchi_town
					duration = -1
				}
			}
		}
		else_if = {
			limit = { owns = 1020 }
			1020 = {
				add_province_modifier = {
					name = jxp_ikko_terauchi_town
					duration = -1
				}
			}
		}
		else = {
			capital_scope = {
				add_province_modifier = {
					name = jxp_ikko_terauchi_town
					duration = -1
				}
			}
		}
		add_adm_power = 25
	}

	option = {
		name = "jxp_ikko.2.b"
		set_country_flag = jxp_ikko_terauchi_seen
		set_country_flag = jxp_ikko_contact
		add_adm_power = -25
		add_stability = 1
		jxp_add_tenka_order_5_effect = yes
	}
}
