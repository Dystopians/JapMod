namespace = jxp_wokou

country_event = {
	id = jxp_wokou.1
	title = "jxp_wokou.1.t"
	desc = "jxp_wokou.1.d"
	picture = NAVAL_MILITARY_eventPicture

	is_triggered_only = yes
	trigger = {
		jxp_is_japanese_polity_trigger = yes
		OR = {
			tag = WAK
			has_country_flag = jxp_path_wokou
			AND = {
				NOT = { jxp_has_any_route_trigger = yes }
				NOT = { has_country_flag = jxp_path_commercial_council }
				jxp_has_wokou_opening_trigger = yes
			}
		}
	}

	option = {
		name = "jxp_wokou.1.a"
		jxp_clear_all_route_flags_effect = yes
		set_country_flag = jxp_path_wokou
		add_country_modifier = {
			name = jxp_route_wokou
			duration = -1
		}
		jxp_grant_route_reforms_effect = yes
		jxp_refresh_route_missions_effect = yes
		jxp_ensure_polity_mechanic_effect = yes
	}
}
