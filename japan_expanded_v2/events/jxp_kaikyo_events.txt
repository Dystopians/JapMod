namespace = jxp_kaikyo

country_event = {
	id = jxp_kaikyo.0
	title = none
	desc = none
	picture = COURT_eventPicture
	hidden = yes

	trigger = {
		jxp_is_japanese_polity_trigger = yes
		has_country_flag = jxp_polity_initialized
		jxp_not_sakoku_locked_trigger = yes
		NOT = { jxp_has_any_route_trigger = yes }
		NOT = { has_country_flag = jxp_kaikyo_contact }
		jxp_oceanic_opening_at_least_50 = yes
		num_of_ports = 5
		OR = {
			is_year = 1550
			any_known_country = { religion_group = muslim }
			any_owned_province = {
				OR = {
					region = malaya_region
					region = indonesia_region
					region = moluccas_region
				}
			}
		}
	}

	mean_time_to_happen = {
		months = 160
	}

	immediate = {
		country_event = { id = jxp_kaikyo.1 days = 1 }
	}

	option = {
		name = "OK"
	}
}
