country_event = {
	id = jxp_confucian_shinto.0
	title = none
	desc = none
	picture = COURT_eventPicture
	hidden = yes

	trigger = {
		jxp_confucian_route_country_trigger = yes
		NOT = { has_country_flag = jxp_confucian_shinto_sync_recent }
		OR = {
			NOT = { has_harmonized_with = shinto }
			NOT = { has_country_flag = jxp_confucian_shinto_syncretism }
			NOT = { has_country_modifier = jxp_confucian_shinto_synthesis }
			NOT = { has_country_flag = jxp_confucian_native_centers_v1 }
			NOT = { jxp_confucian_has_three_native_centers_trigger = yes }
		}
	}

	mean_time_to_happen = {
		days = 1
	}

	immediate = {
		set_country_flag = jxp_confucian_shinto_sync_recent
		jxp_confucian_sync_shinto_effect = yes
		if = {
			limit = {
				OR = {
					NOT = { has_country_flag = jxp_confucian_native_centers_v1 }
					NOT = { jxp_confucian_has_three_native_centers_trigger = yes }
				}
			}
			jxp_spawn_confucian_reform_centers_effect = yes
		}
		jxp_refresh_route_missions_effect = yes
		country_event = { id = jxp_confucian_shinto.99 days = 365 }
	}

	option = {
		name = "OK"
	}
}
