country_event = {
	id = jxp_colonial.1
	title = "jxp_colonial.1.t"
	desc = "jxp_colonial.1.d"
	picture = MERCHANTS_TALKING_eventPicture

	trigger = {
		jxp_is_japanese_polity_trigger = yes
		jxp_can_use_overseas_expansion_trigger = yes
		has_country_flag = jxp_polity_initialized
		is_year = 1620
		NOT = { has_country_flag = jxp_pacific_charter_seen }
		jxp_oceanic_opening_at_least_60 = yes
		num_of_ports = 6
		OR = {
			has_country_flag = jxp_path_open_trade
			has_country_flag = jxp_path_kirishitan
			has_country_flag = jxp_path_reformed
			has_country_flag = jxp_path_kaikyo
			has_country_flag = jxp_path_wokou
			tag = KJP
			tag = RFJ
			tag = SJP
			tag = WAK
			num_of_colonists = 1
		}
	}

	mean_time_to_happen = {
		months = 240
	}

	option = {
		name = "jxp_colonial.1.a"
		set_country_flag = jxp_pacific_charter_seen
		jxp_enact_pacific_charter_effect = yes
	}

	option = {
		name = "jxp_colonial.1.b"
		set_country_flag = jxp_pacific_charter_seen
		set_country_flag = jxp_pacific_charter_enacted
		jxp_add_imperial_sanction_5_effect = yes
		jxp_add_tenka_order_5_effect = yes
		jxp_establish_oceanic_sun_state_effect = yes
		add_country_modifier = {
			name = jxp_oceanic_sun_state
			duration = 3650
		}
	}
}
