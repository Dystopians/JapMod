jxp_shimabara_fire = {
	potential = {
		normal_or_historical_nations = yes
		jxp_is_japanese_polity_trigger = yes
		is_free_or_tributary_trigger = yes
		NOT = { has_country_flag = jxp_shimabara_resolved }
		NOT = { has_country_flag = jxp_path_kirishitan }
		NOT = { has_country_flag = jxp_path_reformed }
		NOT = { tag = KJP }
		NOT = { tag = RFJ }
	}

	can_start = {
		has_any_disaster = no
		is_year = 1620
		OR = {
			NOT = { stability = 1 }
			war_exhaustion = 3
			NOT = { religious_unity = 0.9 }
		}
		OR = {
			has_country_flag = jxp_hidden_christians_seen
			any_owned_province = {
				OR = {
					area = northern_kyushu
					area = kyushu_area
					has_province_flag = jxp_map_compat_shimabara_belt
				}
				OR = {
					religion = catholic
					religion = protestant
					religion = reformed
				}
			}
		}
	}

	can_stop = {
		OR = {
			has_any_disaster = yes
			has_country_flag = jxp_shimabara_resolved
			has_country_flag = jxp_path_kirishitan
			has_country_flag = jxp_path_reformed
			tag = KJP
			tag = RFJ
		}
	}

	progress = {
		modifier = {
			factor = 1
			NOT = { stability = 1 }
		}
		modifier = {
			factor = 2
			NOT = { stability = 0 }
		}
		modifier = {
			factor = 1
			war_exhaustion = 3
		}
		modifier = {
			factor = 1
			NOT = { religious_unity = 0.9 }
		}
		modifier = {
			factor = 2
			has_country_modifier = jxp_expelled_padres
		}
		modifier = {
			factor = -1
			has_country_modifier = jxp_reviewed_translations
		}
		modifier = {
			factor = -1
			has_country_modifier = jxp_dejima_window
		}
		modifier = {
			factor = -1
			has_country_modifier = jxp_shimabara_mediation
		}
	}

	can_end = {
		OR = {
			has_country_flag = jxp_shimabara_resolved
			AND = {
				stability = 1
				NOT = { num_of_rebel_armies = 1 }
				NOT = {
					any_owned_province = {
						OR = {
							area = northern_kyushu
							area = kyushu_area
							has_province_flag = jxp_map_compat_shimabara_belt
						}
						OR = {
							religion = catholic
							religion = protestant
							religion = reformed
						}
					}
				}
			}
		}
	}

	modifier = {
		global_unrest = 4
		tolerance_heretic = -2
		stability_cost_modifier = 0.15
	}

	on_start = jxp_kirishitan.4

	on_end = {
		set_country_flag = jxp_shimabara_resolved
		remove_country_modifier = jxp_shimabara_disaster_modifier
	}
}

jxp_ikko_rising = {
	potential = {
		normal_or_historical_nations = yes
		jxp_is_japanese_polity_trigger = yes
		is_free_or_tributary_trigger = yes
		NOT = { has_country_flag = jxp_ikko_rising_resolved }
		NOT = { jxp_has_any_route_trigger = yes }
		NOT = { tag = IJP }
	}

	can_start = {
		has_any_disaster = no
		is_year = 1470
		num_of_cities = 8
		NOT = { jxp_tenka_order_at_least_50 = yes }
		OR = {
			owns = 1021
			owns = 1020
			any_owned_province = {
				OR = {
					area = kinai_area
					area = saigoku_area
					has_province_flag = jxp_map_compat_ikko_heartland
				}
			}
		}
		OR = {
			NOT = { stability = 1 }
			war_exhaustion = 3
			NOT = { religious_unity = 0.9 }
		}
	}

	can_stop = {
		OR = {
			has_any_disaster = yes
			has_country_flag = jxp_ikko_rising_resolved
			has_country_flag = jxp_path_ikko
			tag = IJP
		}
	}

	progress = {
		modifier = {
			factor = 1
			NOT = { stability = 1 }
		}
		modifier = {
			factor = 1
			war_exhaustion = 3
		}
		modifier = {
			factor = 1
			NOT = { religious_unity = 0.9 }
		}
		modifier = {
			factor = 2
			NOT = { jxp_tenka_order_at_least_25 = yes }
		}
		modifier = {
			factor = -1
			jxp_tenka_order_at_least_50 = yes
		}
		modifier = {
			factor = -1
			has_country_modifier = jxp_ikko_reconciled
		}
	}

	can_end = {
		OR = {
			has_country_flag = jxp_ikko_rising_resolved
			has_country_flag = jxp_path_ikko
			tag = IJP
			AND = {
				stability = 1
				jxp_tenka_order_at_least_50 = yes
			}
		}
	}

	modifier = {
		global_unrest = 3
		stability_cost_modifier = 0.10
	}

	on_start = jxp_ikko.6

	on_end = {
		set_country_flag = jxp_ikko_rising_resolved
		remove_country_modifier = jxp_ikko_disaster_modifier
	}
}
