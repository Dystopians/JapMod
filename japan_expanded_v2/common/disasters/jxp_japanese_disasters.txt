jxp_shimabara_fire = {
	potential = {
		normal_or_historical_nations = yes
		jxp_is_japanese_polity_trigger = yes
		is_free_or_tributary_trigger = yes
		NOT = { has_country_flag = jxp_shimabara_resolved }
		NOT = { has_country_flag = jxp_path_kirishitan }
		NOT = { has_country_flag = jxp_path_reformed }
		NOT = { tag = KJP }
		NOT = { tag = RFJ }
	}

	can_start = {
		has_any_disaster = no
		is_year = 1620
		OR = {
			NOT = { stability = 1 }
			war_exhaustion = 3
			NOT = { religious_unity = 0.9 }
		}
		OR = {
			has_country_flag = jxp_hidden_christians_seen
			any_owned_province = {
				OR = {
					area = northern_kyushu
					area = kyushu_area
					has_province_flag = jxp_map_compat_shimabara_belt
				}
				OR = {
					religion = catholic
					religion = protestant
					religion = reformed
				}
			}
		}
	}

	can_stop = {
		OR = {
			has_any_disaster = yes
			has_country_flag = jxp_shimabara_resolved
			has_country_flag = jxp_path_kirishitan
			has_country_flag = jxp_path_reformed
			tag = KJP
			tag = RFJ
		}
	}

	progress = {
		modifier = {
			factor = 1
			NOT = { stability = 1 }
		}
		modifier = {
			factor = 2
			NOT = { stability = 0 }
		}
		modifier = {
			factor = 1
			war_exhaustion = 3
		}
		modifier = {
			factor = 1
			NOT = { religious_unity = 0.9 }
		}
		modifier = {
			factor = 2
			has_country_modifier = jxp_expelled_padres
		}
		modifier = {
			factor = -1
			has_country_modifier = jxp_reviewed_translations
		}
		modifier = {
			factor = -1
			has_country_modifier = jxp_dejima_window
		}
		modifier = {
			factor = -1
			has_country_modifier = jxp_shimabara_mediation
		}
	}

	can_end = {
		OR = {
			has_country_flag = jxp_shimabara_resolved
			AND = {
				had_country_flag = {
					flag = jxp_shimabara_crisis_active
					days = 365
				}
				stability = 1
				NOT = { num_of_rebel_armies = 1 }
				NOT = {
					any_owned_province = {
						OR = {
							area = northern_kyushu
							area = kyushu_area
							has_province_flag = jxp_map_compat_shimabara_belt
						}
						OR = {
							religion = catholic
							religion = protestant
							religion = reformed
						}
					}
				}
			}
		}
	}

	modifier = {
		global_unrest = 4
		tolerance_heretic = -2
		stability_cost_modifier = 0.15
	}

	on_start = jxp_kirishitan.4

	on_end = jxp_disasters.90
}

# The 1651 Keian uprising was an abortive ronin conspiracy against the
# Tokugawa bakufu.  This alt-history form is deliberately limited to a large,
# unified Japanese monarchy holding Edo and already under two distinct kinds
# of military-fiscal pressure.
jxp_keian_ronin_crisis = {
	potential = {
		normal_or_historical_nations = yes
		jxp_is_japanese_polity_trigger = yes
		is_free_or_tributary_trigger = yes
		jxp_is_unified_japan_state_trigger = yes
		government = monarchy
		NOT = { has_country_flag = jxp_keian_ronin_crisis_resolved }
		NOT = { tag = IJP }
		NOT = { tag = WAK }
	}

	can_start = {
		has_any_disaster = no
		is_year = 1651
		num_of_cities = 20
		owns = 1028
		OR = {
			NOT = { stability = 1 }
			war_exhaustion = 3
		}
		OR = {
			num_of_loans = 2
			NOT = { manpower_percentage = 0.50 }
		}
	}

	can_stop = {
		OR = {
			has_any_disaster = yes
			has_country_flag = jxp_keian_ronin_crisis_resolved
			NOT = { jxp_is_unified_japan_state_trigger = yes }
			NOT = { government = monarchy }
		}
	}

	progress = {
		modifier = {
			factor = 1
			NOT = { stability = 1 }
		}
		modifier = {
			factor = 1
			war_exhaustion = 3
		}
		modifier = {
			factor = 1
			num_of_loans = 2
		}
		modifier = {
			factor = 1
			NOT = { manpower_percentage = 0.50 }
		}
		modifier = {
			factor = -1
			stability = 1
		}
		modifier = {
			factor = -1
			NOT = { num_of_loans = 1 }
		}
		modifier = {
			factor = -2
			has_country_modifier = jxp_keian_ronin_reemployment
		}
		modifier = {
			factor = -1
			has_country_modifier = jxp_keian_ronin_patrols
		}
	}

	can_end = {
		OR = {
			has_country_flag = jxp_keian_ronin_crisis_resolved
			NOT = { jxp_is_unified_japan_state_trigger = yes }
			NOT = { government = monarchy }
			AND = {
				had_country_flag = {
					flag = jxp_keian_ronin_crisis_active
					days = 365
				}
				stability = 1
				NOT = { num_of_loans = 1 }
				manpower_percentage = 0.60
				NOT = { num_of_rebel_armies = 1 }
			}
		}
	}

	modifier = {
		global_unrest = 3
		stability_cost_modifier = 0.10
		global_manpower_modifier = -0.10
	}

	on_start = jxp_disasters.1

	on_end = jxp_disasters.92
}
