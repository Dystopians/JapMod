jxp_76_pacific_network_eligible_trigger = {
	has_country_flag = jxp_polity_initialized
	jxp_allows_private_overseas_trade_trigger = yes
	OR = {
		has_country_flag = jxp_pacific_charter_enacted
		has_country_modifier = jxp_pacific_charter
		mission_completed = jxp_mission_pacific_charter
	}
}

jxp_76_pacific_network_active_trigger = {
	jxp_76_pacific_network_eligible_trigger = yes
	has_country_flag = jxp_76_pacific_network_active
	has_any_disaster = no
}

# Manila-Nagasaki is primarily a merchant, missionary, and maritime route.
jxp_76_pacific_manila_profile_trigger = {
	OR = {
		has_country_flag = jxp_path_open_trade
		has_country_flag = jxp_path_kirishitan
		has_country_flag = jxp_path_reformed
		has_country_flag = jxp_path_kaikyo
		has_country_flag = jxp_path_wokou
		tag = KJP
		tag = RFJ
		tag = SJP
		tag = WAK
	}
}

jxp_76_pacific_manila_anchor_trigger = {
	OR = {
		has_country_flag = jxp_manila_nagasaki_route_seen
		has_country_modifier = jxp_manila_nagasaki_route
		num_of_provinces_owned_or_owned_by_non_sovereign_subjects_with = {
			value = 1
			OR = {
				area = luzon_area
				area = southern_luzon_area
			}
			is_city = yes
		}
	}
}

jxp_76_pacific_alaska_anchor_trigger = {
	is_random_new_world = no
	OR = {
		has_country_flag = jxp_alaska_survey_seen
		has_country_modifier = jxp_north_pacific_charts
		colonial_alaska = { has_discovered = ROOT }
		num_of_provinces_owned_or_owned_by_non_sovereign_subjects_with = {
			value = 1
			colonial_region = colonial_alaska
			is_city = yes
		}
	}
}
