jxp_b_92_regional_colonial_actor_trigger = {
	NOT = { has_country_modifier = jxp_b_97_overseas_rights_revoked }
	OR = {
		AND = {
			has_country_flag = jxp_polity_initialized
			jxp_can_use_overseas_expansion_trigger = yes
		}
		AND = {
			jxp_b_91_colonial_society_active_trigger = yes
			OR = {
				is_colonial_nation = yes
				has_country_flag = jxp_iface_colonial_self_rule
				has_country_flag = jxp_iface_colonial_independence
			}
		}
	}
}

# A closed Japan may maintain one tightly supervised northern exchange, but
# only after it already possesses a controlled foreign channel.
jxp_b_92_northern_colonial_actor_trigger = {
	NOT = { has_country_modifier = jxp_b_97_overseas_rights_revoked }
	OR = {
		jxp_b_92_regional_colonial_actor_trigger = yes
		AND = {
			has_country_flag = jxp_polity_initialized
			jxp_is_japanese_polity_trigger = yes
			jxp_allows_controlled_foreign_channel_trigger = yes
		}
	}
}

# Evaluated in the prospective host-country scope; ROOT is the Japanese party.
jxp_b_92_south_seas_host_candidate_trigger = {
	NOT = { jxp_is_japanese_polity_trigger = yes }
	is_subject = no
	is_at_war = no
	capital_scope = {
		OR = {
			region = indo_china_region
			region = malaya_region
			region = indonesia_region
			region = moluccas_region
		}
	}
	has_opinion = { who = ROOT value = 75 }
	reverse_has_opinion = { who = ROOT value = 25 }
	any_owned_province = {
		jxp_b_92_southeast_asian_port_province_trigger = yes
		NOT = { has_province_modifier = jxp_b_92_authorized_japan_town_port }
	}
}
