	OR = {
		religion = catholic
		religion = protestant
		religion = reformed
		any_owned_province = {
			region = japan_region
			OR = {
				religion = catholic
				religion = protestant
				religion = reformed
			}
		}
		has_country_flag = jxp_kirishitan_contact
		jxp_oceanic_opening_at_least_50 = yes
	}
}

jxp_has_reformed_opening_trigger = {
	OR = {
		religion = protestant

		any_owned_province = {
			region = japan_region
			OR = {
				religion = protestant
				religion = reformed
			}
		}
		has_country_flag = jxp_reformed_contact
		AND = {
			jxp_oceanic_opening_at_least_50 = yes
			dip_tech = 10
			any_known_country = {
				OR = {
					religion = protestant
					religion = reformed
				}
			}
		}
		tag = RFJ
	}
}

jxp_has_confucian_contact_trigger = {
	OR = {
		religion = confucianism
		any_owned_province = {
			OR = {
				religion = confucianism
				region = korea_region
				region = north_china_region
				region = south_china_region
			}
		}
		any_known_country = {
			OR = {
				religion = confucianism
				capital_scope = { region = korea_region }
				capital_scope = { region = north_china_region }
				capital_scope = { region = south_china_region }
			}
		}
		adm_tech = 10
	}
}
