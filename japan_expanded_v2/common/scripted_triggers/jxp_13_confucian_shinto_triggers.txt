jxp_confucian_route_country_trigger = {
	OR = {
		tag = CJP
		has_country_flag = jxp_path_confucian
	}
	religion = confucianism
}

jxp_confucian_shinto_bridge_country_trigger = {
	jxp_confucian_route_country_trigger = yes
	has_harmonized_with = shinto
}

# Once Japan becomes Confucian, completed Shinto incidents keep their vanilla
# end flags. An incident that merely started before conversion remains eligible
# for its Confucian continuation until that end flag is present.
jxp_confucian_bridge_neo_available_trigger = {
	NOT = { has_country_flag = jxp_confucian_bridge_neo_started }
	NOT = { has_country_flag = jxp_confucian_bridge_neo_done }
	NOT = { has_country_flag = end_of_neo_confucianism_chain }
	NOT = { is_incident_active = incident_neo_confucianism }
}

	NOT = { is_incident_active = incident_nanban }
	jxp_allows_controlled_foreign_channel_trigger = yes

		jxp_oceanic_opening_at_least_35 = yes

	}
}

jxp_confucian_has_three_native_centers_trigger = {
	calc_true_if = {
		all_province = {
			religion = confucianism
			is_reformation_center = yes
			has_province_flag = jxp_confucian_native_center
		}
		amount = 3
	}
}
