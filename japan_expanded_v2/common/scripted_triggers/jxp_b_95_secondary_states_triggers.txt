# Formation gates for the three secondary colonial successor states.
# Country/tag registration and central mission/idea refresh remain Agent A interfaces.

jxp_b_95_secondary_colonial_origin_trigger = {
	jxp_b_91_colonial_society_active_trigger = yes
	OR = {
		is_colonial_nation = yes
		has_country_flag = jxp_iface_colonial_self_rule
		has_country_flag = jxp_iface_colonial_independence
		has_country_flag = jxp_b_92_south_seas_japan_town_established
		has_country_flag = jxp_b_92_pacific_island_compact_established
		has_country_flag = jxp_b_92_north_pacific_exchange_established
	}
	NOT = {
		OR = {
			tag = NYA
			tag = HKK
			tag = NJF
			tag = OIA
		}
	}
	NOT = { has_country_flag = jxp_b_95_secondary_state_formed }
}

jxp_b_95_foundation_age_trigger = {
	OR = {
		current_age = age_of_reformation
		current_age = age_of_absolutism
		current_age = age_of_revolutions
		adm_tech = 10
	}
}

jxp_b_95_hkk_normal_heartland_trigger = {
	is_random_new_world = no
	capital_scope = {
		OR = {
			colonial_region = colonial_alaska
			area = kamchatka_area
			area = sakhalin_area
			area = hokkaido_area
		}
	}
	num_of_provinces_owned_or_owned_by_non_sovereign_subjects_with = {
		value = 6
		is_city = yes
		OR = {
			colonial_region = colonial_alaska
			area = kamchatka_area
			area = sakhalin_area
			area = hokkaido_area
		}
	}
	num_of_provinces_owned_or_owned_by_non_sovereign_subjects_with = {
		value = 3
		is_city = yes
		has_port = yes
		OR = {
			colonial_region = colonial_alaska
			area = kamchatka_area
			area = sakhalin_area
			area = hokkaido_area
		}
	}
}

jxp_b_95_hkk_rnw_heartland_trigger = {
	is_random_new_world = yes
	capital_scope = { continent = new_world }
	OR = {
		has_country_flag = jxp_alaska_survey_seen
		has_country_flag = jxp_b_92_north_pacific_exchange_established
		has_country_modifier = jxp_north_pacific_charts
	}
	num_of_provinces_owned_or_owned_by_non_sovereign_subjects_with = {
		value = 6
		continent = new_world
		is_city = yes
	}
	num_of_provinces_owned_or_owned_by_non_sovereign_subjects_with = {
		value = 3
		continent = new_world
		is_city = yes
		has_port = yes
	}
}

jxp_b_95_hkk_heartland_trigger = {
	OR = {
		jxp_b_95_hkk_normal_heartland_trigger = yes
		jxp_b_95_hkk_rnw_heartland_trigger = yes
	}
}
