				region = indonesia_region
				region = moluccas_region
			}
		}
		any_known_country = {
			any_owned_province = {
				has_port = yes

					region = indo_china_region
					region = malaya_region
					region = indonesia_region
					region = moluccas_region
				}
			}
		}
		jxp_oceanic_opening_at_least_40 = yes

}

# Evaluated in the prospective host-country scope. ROOT is the Japanese sponsor.
jxp_b_112_company_host_candidate_trigger = {
	exists = yes
	is_subject = no
	NOT = { tag = ROOT }
	NOT = { is_at_war = yes }
	any_owned_province = {
		has_port = yes
		OR = {
			AND = {
				ROOT = { has_country_flag = jxp_b_112_south_seas_company_active }
				OR = { region = indo_china_region region = malaya_region region = indonesia_region region = moluccas_region }
			}
			AND = {
				ROOT = { has_country_flag = jxp_b_112_north_sea_company_active }
				OR = { area = hokkaido_area area = sakhalin_area area = kamchatka_area colonial_region = colonial_alaska }
			}
			AND = {
				ROOT = { has_country_flag = jxp_b_112_taiwan_fujian_company_active }
				OR = { area = taiwan_area area = fujian_area }
			}
			AND = {
				ROOT = { has_country_flag = jxp_b_112_oceanic_company_active }
				region = oceanea_region
			}
			AND = {
				ROOT = { has_country_flag = jxp_b_112_new_world_company_active }
				OR = {
					continent = north_america
					continent = south_america
					continent = new_world
				}
			}
		}
	}
}
