jxp_b_101_korea_campaign_actor_trigger = {
	jxp_is_japanese_polity_trigger = yes
	is_subject = no
	OR = {
		jxp_is_unified_japan_state_trigger = yes
		AND = {
			has_reform = shogunate
			num_of_cities = 20
		}
	}
}

	KOR = {
		exists = yes
		capital_scope = { region = korea_region }
	}
}

jxp_b_101_can_begin_korea_preparations_trigger = {
	jxp_b_101_korea_campaign_actor_trigger = yes
	jxp_b_101_korea_exists_trigger = yes
	is_at_war = no
	is_year = 1550
	stability = 1
	num_of_cities = 20
	num_of_ports = 5
	naval_forcelimit = 20
	num_of_transport = 8
	manpower_percentage = 0.60
	treasury = 200
	NOT = { is_bankrupt = yes }
	NOT = { has_country_flag = jxp_b_101_korea_campaign_active }
	NOT = { has_country_flag = jxp_b_101_korea_campaign_completed }
}
