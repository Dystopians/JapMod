# A capstone remains reachable after a Japanese final state takes the Mandate.
# The exact final-state gate prevents a stale reform from satisfying another route.
jxp_final_state_uncommitted_power_active_trigger = {
	jxp_final_state_uncommitted_trigger = yes
	OR = {
		AND = { is_emperor_of_china = no has_reform = jxp_uncommitted_realm_council_reform }
		AND = { is_emperor_of_china = yes has_reform = celestial_empire }
	}
}
