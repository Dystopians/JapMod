jxp_b_112_refresh_company_interface_effect = {
	if = {
		limit = { jxp_b_112_has_any_company_trigger = yes }
		set_country_flag = jxp_iface_b_overseas_company_active
	}
	else = {
		clr_country_flag = jxp_iface_b_overseas_company_active
	}
}

jxp_b_112_clear_agent_a_company_requests_effect = {
	clr_country_flag = jxp_iface_a_request_red_seal_overseas_company
	clr_country_flag = jxp_iface_a_request_pacific_charter_company
	clr_country_flag = jxp_iface_a_request_southern_seas_company
	clr_country_flag = jxp_iface_a_request_northern_seas_company
	clr_country_flag = jxp_iface_a_request_continental_company
}

# A owns the paid domestic request. B alone maps it to a regional company,
# checks the remaining capital and route requirements, and clears the request
# after the player accepts or withdraws it.
jxp_b_112_consume_agent_a_company_request_effect = {
	if = {
		limit = { has_country_flag = jxp_iface_a_request_red_seal_overseas_company }
		country_event = { id = jxp_b_company.81 }
	}
	else_if = {
		limit = { has_country_flag = jxp_iface_a_request_pacific_charter_company }
		country_event = { id = jxp_b_company.82 }
	}
	else_if = {
		limit = { has_country_flag = jxp_iface_a_request_southern_seas_company }
		country_event = { id = jxp_b_company.83 }
	}
	else_if = {
		limit = { has_country_flag = jxp_iface_a_request_northern_seas_company }
		country_event = { id = jxp_b_company.84 }
	}
	else_if = {
		limit = { has_country_flag = jxp_iface_a_request_continental_company }
		country_event = { id = jxp_b_company.85 }
	}
}

jxp_b_112_dispatch_company_host_effect = {
	if = {
		limit = { any_known_country = { jxp_b_112_company_host_candidate_trigger = yes } }
		random_known_country = {
			limit = { jxp_b_112_company_host_candidate_trigger = yes }
			if = { limit = { ROOT = { has_country_flag = jxp_b_112_south_seas_company_active } } country_event = { id = jxp_b_company.21 days = 1 } }
			else_if = { limit = { ROOT = { has_country_flag = jxp_b_112_north_sea_company_active } } country_event = { id = jxp_b_company.22 days = 1 } }
			else_if = { limit = { ROOT = { has_country_flag = jxp_b_112_taiwan_fujian_company_active } } country_event = { id = jxp_b_company.23 days = 1 } }
			else_if = { limit = { ROOT = { has_country_flag = jxp_b_112_oceanic_company_active } } country_event = { id = jxp_b_company.24 days = 1 } }
			else_if = { limit = { ROOT = { has_country_flag = jxp_b_112_new_world_company_active } } country_event = { id = jxp_b_company.25 days = 1 } }
		}
	}
	else = {
		set_country_flag = jxp_b_112_company_local_compact
		add_country_modifier = { name = jxp_b_112_company_local_compact duration = 3650 }
	}
}
