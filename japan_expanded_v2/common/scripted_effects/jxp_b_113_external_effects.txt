}

		limit = { NOT = { has_country_flag = jxp_iface_b_china_long_program_active } NOT = { has_country_flag = jxp_iface_b_china_long_program_complete } }
		set_country_flag = jxp_iface_b_china_long_program_active
		clr_country_flag = jxp_b_113_china_stage_1_done
		clr_country_flag = jxp_b_113_china_stage_2_done
		clr_country_flag = jxp_b_113_china_stage_3_done
		clr_country_flag = jxp_b_113_china_stage_4_done
		add_country_modifier = { name = jxp_b_113_china_long_program duration = -1 }
		if = { limit = { has_country_flag = jxp_b_100_china_path_mandate } country_event = { id = jxp_b_china_depth.601 days = 30 } }
		else_if = { limit = { has_country_flag = jxp_b_100_china_path_rites } country_event = { id = jxp_b_china_depth.611 days = 30 } }
		else_if = { limit = { has_country_flag = jxp_b_100_china_path_break } country_event = { id = jxp_b_china_depth.621 days = 30 } }
	}
}
