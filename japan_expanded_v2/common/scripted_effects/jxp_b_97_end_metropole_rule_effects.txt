jxp_b_97_release_filtered_overseas_subjects_effect = {
	every_subject_country = {
		limit = { jxp_b_97_is_overseas_colonial_subject_trigger = yes }
		grant_independence = yes
		set_country_flag = jxp_iface_colonial_self_rule
		set_country_flag = jxp_iface_colonial_independence
		jxp_b_96_clear_independence_crisis_effect = yes
	}
}

jxp_b_97_remove_metropole_overseas_claims_effect = {
	every_province = {
		limit = {
			OR = {
				continent = new_world
				region = oceanea_region
			}
			OR = {
				is_claim = ROOT
				is_permanent_claim = ROOT
			}
		}
		remove_claim = ROOT
	}
}
