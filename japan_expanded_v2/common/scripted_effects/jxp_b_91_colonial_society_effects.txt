jxp_b_91_sync_colonial_stage_effect = {
	jxp_b_91_clamp_colonial_variables_effect = yes
	jxp_b_91_clear_stage_modifiers_effect = yes
	if = {
		limit = { jxp_b_91_colonial_identity_at_least_75_trigger = yes }
		clr_country_flag = jxp_b_colonial_stage_autonomous
		clr_country_flag = jxp_b_colonial_stage_chartered
		add_country_modifier = { name = jxp_b_91_stage_colonial_state duration = -1 }
	}
	else_if = {
		limit = { jxp_b_91_colonial_identity_at_least_50_trigger = yes }
		clr_country_flag = jxp_b_colonial_stage_state
		clr_country_flag = jxp_b_colonial_stage_chartered
		add_country_modifier = { name = jxp_b_91_stage_autonomous_society duration = -1 }
	}
	else_if = {
		limit = { jxp_b_91_colonial_identity_at_least_25_trigger = yes }
		clr_country_flag = jxp_b_colonial_stage_state
		clr_country_flag = jxp_b_colonial_stage_autonomous
		add_country_modifier = { name = jxp_b_91_stage_chartered_settlement duration = -1 }
	}
	else = {
		clr_country_flag = jxp_b_colonial_stage_state
		clr_country_flag = jxp_b_colonial_stage_autonomous
		clr_country_flag = jxp_b_colonial_stage_chartered
		add_country_modifier = { name = jxp_b_91_stage_anchorage duration = -1 }
	}
}
