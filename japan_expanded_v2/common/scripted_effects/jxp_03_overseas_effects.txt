jxp_establish_oceanic_sun_state_effect = {
	set_country_flag = jxp_oceanic_sun_state
	jxp_ensure_polity_mechanic_effect = yes
	if = {
		limit = { has_reform = jxp_oceanic_sun_reform }
		remove_government_reform = jxp_oceanic_sun_reform
	}
	regenerate_government_mechanics = yes
	if = {
		limit = { NOT = { has_country_modifier = jxp_oceanic_sun_state } }
		add_country_modifier = {
			name = jxp_oceanic_sun_state
			duration = -1
		}
	}
}

jxp_apply_mandate_claim_reward_effect = {
	jxp_b_100_choose_mandate_path_effect = yes
	set_country_flag = jxp_mandate_claimed
	clr_country_flag = jxp_mandate_broken
	jxp_add_imperial_sanction_10_effect = yes
	jxp_add_tenka_order_5_effect = yes
	if = {
		limit = {
			has_dlc = "Mandate of Heaven"
			is_emperor_of_china = yes
		}
		add_mandate = 10
	}
	else = {
		set_country_flag = jxp_mandate_claim_fallback
		if = {
			limit = { NOT = { has_country_modifier = jxp_celestial_diplomacy } }
			add_country_modifier = {
				name = jxp_celestial_diplomacy
				duration = 7300
			}
		}
		if = {
			limit = { NOT = { has_country_modifier = jxp_solar_court_recognition } }
			add_country_modifier = {
				name = jxp_solar_court_recognition
				duration = 7300
			}
		}
	}
}
