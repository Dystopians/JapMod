jxp_a_reconcile_oda_government_effect = {
	hidden_effect = {
		if = {
			limit = {
				OR = {
					NOT = { tag = ODA }
					NOT = { jxp_is_daimyo_stage_trigger = yes }
				}
			}
			if = {
				limit = { has_reform = jxp_a_oda_azuchi_magistracy_reform }
				remove_government_reform = jxp_a_oda_azuchi_magistracy_reform
			}
			if = {
				limit = { has_reform = jxp_a_oda_legion_domains_reform }
				remove_government_reform = jxp_a_oda_legion_domains_reform
			}
			if = {
				limit = { has_reform = jxp_a_oda_ashikaga_protectorate_reform }
				remove_government_reform = jxp_a_oda_ashikaga_protectorate_reform
			}
			jxp_a_clear_oda_government_choice_effect = yes
			jxp_a_clear_oda_succession_choice_effect = yes
			clr_country_flag = jxp_a_oda_government_terminal_unlocked
			clr_country_flag = jxp_a_oda_legion_autonomy
			clr_country_flag = jxp_a_oda_direct_magistrates
		}
		else = {
			if = {
				limit = {
					has_reform = jxp_a_oda_azuchi_magistracy_reform
					NOT = { has_country_flag = jxp_a_oda_government_azuchi }
				}
				remove_government_reform = jxp_a_oda_azuchi_magistracy_reform
			}
			if = {
				limit = {
					has_reform = jxp_a_oda_legion_domains_reform
					NOT = { has_country_flag = jxp_a_oda_government_legion_domains }
				}
				remove_government_reform = jxp_a_oda_legion_domains_reform
			}
			if = {
				limit = {
					has_reform = jxp_a_oda_ashikaga_protectorate_reform
					NOT = { has_country_flag = jxp_a_oda_government_ashikaga_protectorate }
				}
				remove_government_reform = jxp_a_oda_ashikaga_protectorate_reform
			}
		}
	}
}
