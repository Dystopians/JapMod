		limit = { NOT = { has_country_flag = jxp_a_shinshu_initialized } }
		set_country_flag = jxp_a_shinshu_initialized
		set_variable = { which = jxp_a_shinshu_militancy value = 20 }

			limit = { has_country_flag = jxp_map_origin_hng }

jxp_a_shinshu_sync_identity_effect = {
	if = {
		limit = {
			OR = {
				tag = IJP
				has_country_flag = jxp_path_ikko
				has_country_flag = jxp_map_origin_hng
			}
			NOT = { religion = jodo_shinshu }
		}
		change_religion = jodo_shinshu
	}
	if = {
		limit = { religion = jodo_shinshu }
		set_ruler_religion = jodo_shinshu
		if = { limit = { has_heir = yes } set_heir_religion = jodo_shinshu }
		capital_scope = { change_religion = jodo_shinshu }
		every_owned_province = {
			limit = {
				OR = {
					province_id = 1021
					province_id = 1023
					has_province_flag = jxp_map_compat_ikko_formable_heartland
					has_province_modifier = jxp_ikko_terauchi_town
				}
			}
			change_religion = jodo_shinshu
		}
	}
	jxp_a_shinshu_initialize_effect = yes
}

jxp_a_shinshu_migrate_v029_effect = {
	if = {
		limit = {
			OR = {
				NOT = { has_country_flag = jxp_a_shinshu_migration_v029 }
				NOT = { jxp_a_shinshu_migration_postcondition_trigger = yes }
			}
		}
		clr_country_flag = jxp_a_shinshu_migration_v029
		if = {
			limit = { jxp_a_shinshu_country_trigger = yes }
			if = {
				limit = { tag = IJP }
				jxp_74_ijp_enforce_route_exclusivity_effect = yes
			}
			jxp_a_shinshu_sync_identity_effect = yes
			if = {
				limit = { check_variable = { which = jxp_74_ijp_pressure value = 85 } }
				set_variable = { which = jxp_a_shinshu_militancy value = 85 }
			}
			else_if = {
				limit = { check_variable = { which = jxp_74_ijp_pressure value = 70 } }
				set_variable = { which = jxp_a_shinshu_militancy value = 70 }
			}
			else_if = {
				limit = { check_variable = { which = jxp_74_ijp_pressure value = 50 } }
				set_variable = { which = jxp_a_shinshu_militancy value = 50 }
			}
			else_if = {
				limit = { check_variable = { which = jxp_74_ijp_pressure value = 10 } }
				set_variable = { which = jxp_a_shinshu_militancy value = 20 }
			}
			set_variable = { which = jxp_74_ijp_pressure value = 0 }
			if = {
				limit = {
					OR = {
						has_disaster = jxp_ikko_rising
						has_country_flag = jxp_ikko_rising_active
					}
				}
				set_country_flag = jxp_ikko_rising_active
				jxp_a_shinshu_set_stage_4_effect = yes
			}
		}
		if = {
			limit = { jxp_a_shinshu_migration_postcondition_trigger = yes }
			set_country_flag = jxp_a_shinshu_migration_v029
		}
	}
}
