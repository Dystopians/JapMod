jxp_clamp_polity_values_effect = {
	hidden_effect = {
		if = {
			limit = { NOT = { check_variable = { which = jxp_tenka_order value = 0 } } }
			set_variable = { which = jxp_tenka_order value = 0 }
		}
		if = {
			limit = { check_variable = { which = jxp_tenka_order value = 100 } }
			set_variable = { which = jxp_tenka_order value = 100 }
		}
		if = {
			limit = { NOT = { check_variable = { which = jxp_imperial_sanction value = 0 } } }
			set_variable = { which = jxp_imperial_sanction value = 0 }
		}
		if = {
			limit = { check_variable = { which = jxp_imperial_sanction value = 100 } }
			set_variable = { which = jxp_imperial_sanction value = 100 }
		}
		if = {
			limit = { NOT = { check_variable = { which = jxp_oceanic_opening value = 0 } } }
			set_variable = { which = jxp_oceanic_opening value = 0 }
		}
		if = {
			limit = { check_variable = { which = jxp_oceanic_opening value = 100 } }
			set_variable = { which = jxp_oceanic_opening value = 100 }
		}
	}
}

jxp_ensure_polity_mechanic_effect = {
	hidden_effect = {
		if = {
			limit = { has_reform = jxp_japanese_polity_reform }
			remove_government_reform = jxp_japanese_polity_reform
		}
		regenerate_government_mechanics = yes
		if = {
			limit = { has_government_mechanic = jxp_japanese_polity_mechanic }
			set_country_flag = jxp_polity_mechanic_registered
			jxp_sync_polity_mechanic_from_variables_effect = yes
		}
		else = {
			clr_country_flag = jxp_polity_mechanic_registered
		}
	}
}

jxp_init_polity_values_effect = {
	set_country_flag = jxp_polity_initialized
	jxp_record_daimyo_origin_effect = yes
	set_variable = { which = jxp_tenka_order value = 25 }
	set_variable = { which = jxp_imperial_sanction value = 35 }
	set_variable = { which = jxp_oceanic_opening value = 20 }
	if = {
		limit = {
			OR = {
				tag = JAP
				has_reform = shogunate
				owns = 1020
			}
		}
		set_variable = { which = jxp_imperial_sanction value = 50 }
	}
	if = {
		limit = {
			OR = {
				tag = KJP
				has_country_flag = jxp_path_kirishitan
				tag = RFJ
				tag = SJP
				tag = WAK
				has_country_flag = jxp_path_reformed
				has_country_flag = jxp_path_kaikyo
				has_country_flag = jxp_path_wokou
			}
		}
		set_variable = { which = jxp_oceanic_opening value = 45 }
	}
	if = {
		limit = {
			OR = {
				tag = IJP
				has_country_flag = jxp_path_ikko
			}
		}
		set_variable = { which = jxp_tenka_order value = 20 }
		set_variable = { which = jxp_imperial_sanction value = 40 }
	}
	jxp_clamp_polity_values_effect = yes
	jxp_ensure_polity_mechanic_effect = yes
}
