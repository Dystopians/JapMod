jxp_b_100_choose_mandate_path_effect = {
	if = {
		limit = { NOT = { jxp_b_100_has_any_china_endgame_path_trigger = yes } }
		set_country_flag = jxp_b_100_china_path_mandate
		set_country_flag = jxp_mandate_claimed
		clr_country_flag = jxp_mandate_broken
		add_country_modifier = {
			name = jxp_b_100_mandate_proclamation
			duration = 7300
		}
		jxp_b_113_start_china_long_program_effect = yes
	}
}

jxp_b_100_accept_b99_china_campaign_effect = {
	if = {
		limit = { has_country_flag = jxp_iface_b99_china_campaign_requested }
		clr_country_flag = jxp_iface_b99_china_campaign_requested
		set_country_flag = jxp_b_100_b99_direct_campaign_authorized
		if = {
			limit = { NOT = { jxp_b_100_has_any_china_endgame_path_trigger = yes } }
			jxp_b_100_choose_break_path_effect = yes
		}
	}
}

jxp_b_100_complete_mandate_endgame_effect = {
	if = {
		limit = { jxp_b_100_mandate_endgame_ready_trigger = yes }
		set_country_flag = jxp_b_100_china_endgame_mandate_complete
		add_prestige = 20
		if = {
			limit = {
				has_dlc = "Mandate of Heaven"
				is_emperor_of_china = yes
			}
			add_mandate = 5
			add_country_modifier = {
				name = jxp_b_100_mandate_endgame
				duration = -1
			}
		}
		else = {
			add_country_modifier = {
				name = jxp_b_100_mandate_fallback_endgame
				duration = -1
			}
		}
	}
}

jxp_b_100_complete_rites_endgame_effect = {

jxp_b_100_consolidate_china_endgame_effect = {
	if = {
		limit = { has_country_flag = jxp_b_100_china_endgame_mandate_complete }
		add_legitimacy = 10
		add_dip_power = 50
	}
	else_if = {
		limit = { has_country_flag = jxp_b_100_china_endgame_rites_complete }
		add_mercantilism = 1
		add_dip_power = 50
	}
	else_if = {
		limit = { has_country_flag = jxp_b_100_china_endgame_break_complete }
		add_navy_tradition = 5
		add_dip_power = 50
	}
}
