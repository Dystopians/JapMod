}

jxp_reclassify_founder_house_reforms_v0234_effect = {
	if = {
		limit = { jxp_is_unified_japan_state_trigger = yes }
		if = {
			limit = { jxp_has_any_founder_house_reform_trigger = yes }
			jxp_clear_founder_house_reforms_effect = yes
			add_government_reform = quash_noble_power_reform
			change_government_reform_progress = 100
		}
		clr_country_flag = jxp_founder_reform_assigned_v0232
		jxp_unlock_founder_house_reform_effect = yes
	}
	set_country_flag = jxp_founder_reform_reclassified_v0234
}
