jxp_ensure_polity_for_shogunate_policy_effect = {
	if = {
		limit = { NOT = { has_country_flag = jxp_polity_initialized } }
		jxp_init_polity_values_effect = yes
	}
	if = {
		limit = { has_country_flag = jxp_polity_initialized }
		jxp_ensure_polity_mechanic_effect = yes
	}
}

jxp_apply_shogunate_expel_ronin_effect = {
	jxp_ensure_polity_for_shogunate_policy_effect = yes
	jxp_add_tenka_order_5_effect = yes
}
