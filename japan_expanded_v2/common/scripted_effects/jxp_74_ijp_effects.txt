jxp_74_ijp_clamp_pressure_effect = {
	hidden_effect = {
		if = {
			limit = {
				NOT = { check_variable = { which = jxp_74_ijp_pressure value = 0 } }
			}
			set_variable = { which = jxp_74_ijp_pressure value = 0 }
		}
		if = {
			limit = {
				check_variable = { which = jxp_74_ijp_pressure value = 100 }
			}
			set_variable = { which = jxp_74_ijp_pressure value = 100 }
		}
	}
}
