jxp_76_pacific_clamp_pressure_effect = {
	hidden_effect = {
		if = {
			limit = {
				NOT = { check_variable = { which = jxp_76_pacific_pressure value = 0 } }
			}
			set_variable = { which = jxp_76_pacific_pressure value = 0 }
		}
		if = {
			limit = {
				check_variable = { which = jxp_76_pacific_pressure value = 100 }
			}
			set_variable = { which = jxp_76_pacific_pressure value = 100 }
		}
	}
}
