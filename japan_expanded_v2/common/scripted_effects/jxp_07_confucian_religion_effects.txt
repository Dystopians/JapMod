jxp_confucian_protect_ise_effect = {
	4359 = {
		if = {
			limit = { always = yes }
			if = {
				limit = {
					is_reformation_center = yes
					religion = confucianism
				}
				remove_reform_center = confucianism
			}
			if = {
				limit = { NOT = { religion = shinto } }
				change_religion = shinto
			}
			if = {
				limit = { NOT = { has_province_modifier = jxp_ise_jingu_protected } }
				add_province_modifier = {
					name = jxp_ise_jingu_protected
					duration = -1
				}
			}
			if = {
				limit = { NOT = { has_province_modifier = religious_zeal_at_conv } }
				add_permanent_province_modifier = {
					name = religious_zeal_at_conv
					duration = -1
				}
				set_province_flag = jxp_ise_jingu_zeal_by_jxp
			}
		}
	}
}

jxp_confucian_sync_shinto_effect = {
	if = {
		limit = {
			jxp_confucian_route_country_trigger = yes
			NOT = { has_harmonized_with = shinto }
		}
		add_harmonized_religion = shinto
	}
	if = {
		limit = {
			jxp_confucian_route_country_trigger = yes
			has_harmonized_with = shinto
		}
		set_country_flag = decided_on_confucian_flag
		set_country_flag = jxp_confucian_shinto_syncretism
		if = {
			limit = {
				NOT = { has_country_modifier = jxp_confucian_shinto_synthesis }
			}
			add_country_modifier = {
				name = jxp_confucian_shinto_synthesis
				duration = -1
			}
		}
		set_country_flag = jxp_confucian_shinto_synthesis_granted
	}
	jxp_confucian_protect_ise_effect = yes
}

jxp_add_confucian_reform_center_here_effect = {
	if = {
		limit = {
			region = japan_region
			NOT = { province_id = 4359 }
			NOT = { is_reformation_center = yes }
			OR = {
				religion = shinto
				religion = confucianism
			}
			ROOT = {
				jxp_confucian_route_country_trigger = yes
				NOT = { jxp_confucian_has_three_native_centers_trigger = yes }
			}
		}
		change_religion = confucianism
		add_reform_center = confucianism
		set_province_flag = jxp_confucian_native_center
		add_province_modifier = {
			name = jxp_confucian_reform_center
			duration = -1
		}
		add_permanent_province_modifier = {
			name = religious_zeal_at_conv
			duration = 9000
		}
	}
}

jxp_spawn_confucian_reform_centers_effect = {
	hidden_effect = {
		if = {
			limit = {
				NOT = { has_country_flag = jxp_confucian_native_centers_v1 }
			}
			every_owned_province = {
				limit = {
					region = japan_region
					NOT = { province_id = 4359 }
					has_province_modifier = jxp_confucian_reform_center
					NOT = { is_reformation_center = yes }
					OR = {
						religion = shinto
						religion = confucianism
					}
					ROOT = {
						NOT = { jxp_confucian_has_three_native_centers_trigger = yes }
					}
				}
				jxp_add_confucian_reform_center_here_effect = yes
			}
		}

		if = {
			limit = {
				owns = 1020
				1020 = {
					NOT = { is_reformation_center = yes }
					OR = {
						religion = shinto
						religion = confucianism
					}
				}
			}
			1020 = { jxp_add_confucian_reform_center_here_effect = yes }
		}
		if = {
			limit = {
				owns = 4182
				4182 = {
					NOT = { is_reformation_center = yes }
					OR = {
						religion = shinto
						religion = confucianism
					}
				}
			}
			4182 = { jxp_add_confucian_reform_center_here_effect = yes }
		}
		if = {
			limit = {
				owns = 1028
				1028 = {
					NOT = { is_reformation_center = yes }
					OR = {
						religion = shinto
						religion = confucianism
					}
				}
			}
			1028 = { jxp_add_confucian_reform_center_here_effect = yes }
		}

		capital_scope = {
			if = {
				limit = {
					region = japan_region
					NOT = { province_id = 4359 }
					NOT = { is_reformation_center = yes }
					OR = {
						religion = shinto
						religion = confucianism
					}
				}
				jxp_add_confucian_reform_center_here_effect = yes
			}
		}

		if = {
			limit = {
				NOT = { jxp_confucian_has_three_native_centers_trigger = yes }
				any_owned_province = {
					region = japan_region
					NOT = { province_id = 4359 }
					NOT = { is_reformation_center = yes }
					OR = {
						religion = shinto
						religion = confucianism
					}
				}
			}
			random_owned_province = {
				limit = {
					region = japan_region
					NOT = { province_id = 4359 }
					NOT = { is_reformation_center = yes }
					OR = {
						religion = shinto
						religion = confucianism
					}
				}
				jxp_add_confucian_reform_center_here_effect = yes
			}
		}
		if = {
			limit = {
				NOT = { jxp_confucian_has_three_native_centers_trigger = yes }
				any_owned_province = {
					region = japan_region
					NOT = { province_id = 4359 }
					NOT = { is_reformation_center = yes }
					OR = {
						religion = shinto
						religion = confucianism
					}
				}
			}
			random_owned_province = {
				limit = {
					region = japan_region
					NOT = { province_id = 4359 }
					NOT = { is_reformation_center = yes }
					OR = {
						religion = shinto
						religion = confucianism
					}
				}
				jxp_add_confucian_reform_center_here_effect = yes
			}
		}
		if = {
			limit = {
				NOT = { jxp_confucian_has_three_native_centers_trigger = yes }
				any_owned_province = {
					region = japan_region
					NOT = { province_id = 4359 }
					NOT = { is_reformation_center = yes }
					OR = {
						religion = shinto
						religion = confucianism
					}
				}
			}
			random_owned_province = {
				limit = {
					region = japan_region
					NOT = { province_id = 4359 }
					NOT = { is_reformation_center = yes }
					OR = {
						religion = shinto
						religion = confucianism
					}
				}
				jxp_add_confucian_reform_center_here_effect = yes
			}
		}

		every_province = {
			limit = {
				has_province_modifier = jxp_confucian_reform_center
				NOT = { has_province_flag = jxp_confucian_native_center }
			}
			remove_province_modifier = jxp_confucian_reform_center
		}

		set_country_flag = jxp_confucian_native_centers_v1
		set_country_flag = jxp_confucian_centers_established
		clr_country_flag = jxp_confucian_conversion_active
		jxp_confucian_protect_ise_effect = yes
	}
}

jxp_confucian_convert_one_japan_province_effect = {
	hidden_effect = {
		if = {
			limit = {
				jxp_confucian_route_country_trigger = yes
				NOT = { has_country_flag = jxp_confucian_native_centers_v1 }
			}
			jxp_spawn_confucian_reform_centers_effect = yes
		}
	}
}

jxp_remove_confucian_native_centers_effect = {
	hidden_effect = {
		every_province = {
			limit = { has_province_flag = jxp_confucian_native_center }
			if = {
				limit = { is_reformation_center = yes }
				remove_reform_center = confucianism
			}
			remove_province_modifier = jxp_confucian_reform_center
			clr_province_flag = jxp_confucian_native_center
		}
	}
}
