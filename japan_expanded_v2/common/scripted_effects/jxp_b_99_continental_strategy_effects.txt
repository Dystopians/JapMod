		add_country_modifier = { name = jxp_b_99_doctrine_imperial duration = 3650 }

		set_country_flag = jxp_b_99_campaign_china
		set_country_flag = jxp_iface_b99_china_campaign_requested

		event_target:jxp_b_99_theater_target = { add_country_modifier = { name = jxp_b_99_alarm_southeast_asia duration = 3650 } }
	}
	add_country_modifier = { name = jxp_b_99_campaign_preparations duration = 3650 }

		limit = {
			NOT = {
				OR = {
