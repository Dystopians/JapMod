	valid_for_subject = no

	prerequisites_self = {
		jxp_b_100_can_use_mandate_acquisition_cb_trigger = yes

			ai = no
			jxp_b_118_ai_mandate_war_ready_trigger = yes
		}
	}

		FROM = {
			is_emperor_of_china = yes

	}

}
